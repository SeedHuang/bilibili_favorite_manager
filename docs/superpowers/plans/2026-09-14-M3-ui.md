```ts
import type Database from 'better-sqlite3';
import type { Logger } from '../logger/index.js';

export interface HttpDeps {
  db: Database.Database;
  log: Logger;
}

export function createServer(deps: HttpDeps): FastifyInstance {
  const { db, log } = deps;
  const app = Fastify({ logger: false }); // 用我们自己的 Logger,不叠 pino

  app.get('/api/health', async () => {
    const folders = (db.prepare(`SELECT COUNT(*) AS n FROM folders`).get() as { n: number }).n;
    const items = (db.prepare(`SELECT COUNT(*) AS n FROM items`).get() as { n: number }).n;
    return { ok: true, folders, items };
  });

  return app;
}

/** 供开发/测试:启动到某端口 */
export async function startServer(
  deps: HttpDeps,
  port = 3001,
): Promise<FastifyInstance> {
  const app = createServer(deps);
  await app.listen({ port, host: '127.0.0.1' });
  return app;
}
```

`server/src/security/dpapi.ts`:

```ts
import { Dpapi, isPlatformSupported } from '@primno/dpapi';

const SCOPE = 'CurrentUser' as const;
const enc = new TextEncoder();
const dec = new TextDecoder();

/** DPAPI 在当前平台可用吗?(仅 Windows x64/arm64) */
export function dpapiAvailable(): boolean {
  return isPlatformSupported;
}

/** 加密为 base64。非 Windows 平台降级为明文(标注前缀以便识别) */
export function encryptSecret(plain: string): string {
  if (!dpapiAvailable()) return `plain:${plain}`;
  const encBuf = Dpapi.protectData(enc.encode(plain), null, SCOPE);
  return Buffer.from(encBuf).toString('base64');
}

/** 解密。非 Windows / 解不出 → null */
export function decryptSecret(stored: string): string | null {
  if (stored.startsWith('plain:')) return stored.slice('plain:'.length);
  if (!dpapiAvailable()) return null;
  try {
    const buf = Buffer.from(stored, 'base64');
    const out = Dpapi.unprotectData(new Uint8Array(buf), null, SCOPE);
    return dec.decode(out);
  } catch {
    return null; // 换机器/换账户/损坏 → 解不出,调用方重新授权
  }
}
```

```ts
import type { FastifyInstance } from 'fastify';
import type Database from 'better-sqlite3';
import type { Logger } from '../../logger/index.js';
import type { BiliClient } from '../../bilibili/client.js';
import { RiskControlError, AuthError } from '../../bilibili/errors.js';
import { getSetting, setSetting } from '../../db/repo/state.js';
import { encryptSecret } from '../../security/dpapi.js';

export interface AuthDeps {
  db: Database.Database;
  log: Logger;
  /** 用于验证 cookie 的客户端 */
  client: BiliClient;
}

export function registerAuthRoutes(app: FastifyInstance, deps: AuthDeps): void {
  const { db, log, client } = deps;

  app.post('/api/auth/validate', async (req, reply) => {
    const { cookie } = req.body as { cookie?: string };
    if (!cookie || typeof cookie !== 'string' || !cookie.trim()) {
      return reply.code(400).send({ ok: false, reason: '没有收到 cookie' });
    }

    // 从整段粘贴里提取凭证(复用 M2 的纯函数)
    const { sessdata, bilijct } = extractCreds(cookie);
    if (!sessdata || !bilijct) {
      return reply.code(400).send({
        ok: false,
        reason: '没能从粘贴内容里同时找到 SESSDATA 和 bili_jct —— 检查是否复制完整',
      });
    }

    // 用用户刚粘贴的凭证构造一个临时客户端,真实调 nav 验证。
    // 复用 app 里已有客户端(它已带指纹和限速器),只是换掉凭证 ——
    // 通过 BiliClient 新增的公开工厂方法做,避免索引访问私有字段。
    const probe = client.withCredentials(sessdata, bilijct);

    try {
      const nav = await probe.get<{ mid?: number; uname?: string }>('/x/web-interface/nav');
      if (!nav?.mid) {
        return reply.code(401).send({ ok: false, reason: 'cookie 已过期或登录态不完整,请重新登录后复制' });
      }

      // 加密存储(C10)
      setSetting(db, 'bili.sessdata', encryptSecret(sessdata));
      setSetting(db, 'bili.bilijct', encryptSecret(bilijct));

      log.event({
        level: 'info', category: 'auth',
        message: `授权成功:${nav.uname} (UID ${nav.mid})`,
      });

      return { ok: true, mid: nav.mid, uname: nav.uname ?? '' };
    } catch (e) {

      }
      return reply.code(502).send({ ok: false, reason: `验证失败:${(e as Error)?.message ?? e}` });
    }
  });
}
```

```ts
import type { FastifyInstance } from 'fastify';
import type Database from 'better-sqlite3';
import type { Logger } from '../../logger/index.js';
import { getItem, listItemIdsByFolder } from '../../db/repo/items.js';

/** 建 FTS5 虚拟表(幂等)+ 同步触发器 */
function ensureFts(db: Database.Database): void {
  db.exec(`
    CREATE VIRTUAL TABLE IF NOT EXISTS items_fts USING fts5(
      id UNINDEXED, title, intro, upper_name,
      content='items', content_rowid='rowid'
    );
    -- 简单触发器:INSERT/UPDATE/DELETE 时同步 FTS(用 external content 表更省,但先求正确)
  `);
}

export function registerItemRoutes(app: FastifyInstance, deps: { db: Database.Database; log: Logger }): void {
  const { db, log } = deps;
  ensureFts(db);

  app.get('/api/folders/:id/items', async (req) => {
    const { id } = req.params as { id: string };
    const page = Number((req.query as { page?: string }).page ?? 1);
    const pageSize = Number((req.query as { pageSize?: string }).pageSize ?? 20);
    const ids = listItemIdsByFolder(db, Number(id), { limit: pageSize, offset: (page - 1) * pageSize });
    const items = ids.map((iid) => getItem(db, iid)).filter(Boolean);
    const total = ids.length > 0 || page === 1 ? (() => {
      const r = db.prepare(`SELECT COUNT(*) AS n FROM folder_items WHERE folder_id = ?`).get(Number(id)) as { n: number };
      return r.n;
    })() : 0;
    return { items: items.map(shape), total };
  });

  // ... search 与 detail 类似
}

function shape(i: NonNullable<ReturnType<typeof getItem>>) {
  return {
    id: i.id, title: i.title, cover: i.cover, duration: i.duration,
    pubtime: i.pubtime, favTime: i.fav_time, upperName: i.upper_name, invalid: i.invalid,
  };
}
```

> **注意**:FTS 的触发器是 M3 的正确做法,但 `ensureFts` 里的触发器只覆盖了"本进程内新增"。
> M2 的同步是独立进程跑的(`sync-live.ts`),**那不会经过 HTTP 层的 ensureFts**。所以 FTS 索引
> 需要在**同步完成后再重建**(`INSERT INTO items_fts(items_fts) VALUES('rebuild')`)。
> 这个重建的时机在 M3 里交给「同步完成后调用」—— 但同步本身是 M3 之外的。
> **简化方案**:搜索路由每次先 `rebuild`(数据量 3000 条,rebuild 毫秒级,可接受)。

```ts
import type { FastifyInstance } from 'fastify';
import type Database from 'better-sqlite3';
import type { Logger } from '../../logger/index.js';

export function registerSseRoutes(app: FastifyInstance, deps: { db: Database.Database; log: Logger }): void {
  const { db } = deps;

  app.get('/api/events', async (req, reply) => {
    reply.raw.writeHead(200, {
      'Content-Type': 'text/event-stream',
      'Cache-Control': 'no-cache',
      Connection: 'keep-alive',
    });

    // last-event-id 续传:从该 id 之后开始
    const lastId = Number(req.headers['last-event-id'] ?? 0) || 0;
    const pollEvents = (): void => {
      const rows = db.prepare(`SELECT id, level, category, code, message FROM events WHERE id > ? ORDER BY id LIMIT 20`).all(lastId) as
        Array<{ id: number; level: string; category: string; code: string | null; message: string }>;
      for (const row of rows) {
        reply.raw.write(`id: ${row.id}\nevent: log\ndata: ${JSON.stringify(row)}\n\n`);
        lastId = row.id;
      }
    };

    pollEvents();
    // 简单轮询(1s):每轮把新事件推出去
    const timer = setInterval(pollEvents, 1000);
    req.raw.on('close', () => clearInterval(timer));
    // 测试钩子:?test=1 时写完首批后主动结束,让 app.inject 能返回
    if ((req.query as { test?: string }).test === '1') {
      pollEvents();
      clearInterval(timer);
      reply.raw.end();
    }
  });
}
```

> **注意**:上面的 `lastId` 必须用 `let`,因为在循环里更新。

> **测试的关键坑(已定)**:`app.inject()` 对 SSE 会等流结束才返回,而 setInterval 让流永不结束,
> 测试会挂起。**已定方案**:给 SSE 路由加一个 `?test=1` 查询参数 —— 写完全部已排队数据后
> `clearInterval` + `reply.raw.end()`,让测试能干净返回。正常路径(无 test=1)保持长连接轮询。
> 两个测试都带 `?test=1`。**注意**:上面的实现里 test 分支调了 `pollEvents()` 后立即 end ——
> 但 test=1 时首次连接还没有事件,`pollEvents` 在第一次连接时把历史事件推出去,测试靠
> `last-event-id: 0` 拿全部,能覆盖"收到已排队事件"的断言。`test=1` 的正确语义:写完当下
> 已排队的数据后结束,不再开新定时器。

```ts
/** 薄封装:所有 /api 请求。CSRF 对本机工具无意义,不处理。 */
export async function api<T>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(path, init);
  if (!res.ok) {
    const body = await res.json().catch(() => ({}));
    throw new Error((body as { reason?: string }).reason ?? `请求失败 ${res.status}`);
  }
  return res.json() as Promise<T>;
}
```

- [ ] **Step 2: 写 auth 页**

`web/src/pages/auth.tsx`:

```tsx
import { useState } from 'react';
import { Button, Input, Alert, Spin } from 'antd';
import { CheckCircleOutlined } from 'lucide-react';
import { api } from '../api';

export default function AuthPage() {
  const [cookie, setCookie] = useState('');
  const [status, setStatus] = useState<'idle' | 'loading' | 'done' | 'error'>('idle');
  const [account, setAccount] = useState<{ mid: number; uname: string } | null>(null);
  const [error, setError] = useState('');

  async function handleValidate() {
    setStatus('loading');
    try {
      const r = await api<{ ok: true; mid: number; uname: string }>('/api/auth/validate', {
        method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ cookie }),
      });
      setAccount({ mid: r.mid, uname: r.uname });
      setStatus('done');
    } catch (e) {
      setError((e as Error).message);
      setStatus('error');
    }
  }

  if (status === 'done' && account) {
    return (
      <div>
        <CheckCircleOutlined style={{ color: 'var(--ok)' }} />
        <h2>已连接</h2>
        <p>● {account.uname} &nbsp; UID:{account.mid}</p>
        <Button onClick={() => { setStatus('idle'); setAccount(null); }}>重新授权</Button>
      </div>
    );
  }

  return (
    <div style={{ maxWidth: 640 }}>
      <h1>授权</h1>
      <p>粘贴你的 bilibili cookie(整段 Copy as cURL 或裸 cookie 串均可)。保存前会验证登录态。</p>
      <Input.TextArea
        rows={6}
        value={cookie}
        onChange={(e) => setCookie(e.target.value)}
        placeholder="SESSDATA=...; bili_jct=... 或整段 cURL"
        style={{ fontFamily: 'monospace' }}
      />
      <div style={{ marginTop: 12 }}>
        <Button type="primary" onClick={handleValidate} disabled={!cookie.trim()} loading={status === 'loading'}>
          验证并保存
        </Button>
      </div>
      {status === 'error' && <Alert type="error" message={error} style={{ marginTop: 12 }} />}
    </div>
  );
}
```

`web/src/pages/index.tsx`:

```tsx
import { useState } from 'react';
import { useRequest } from '@umijs/max';
import { Search } from 'lucide-react';
import { Input } from 'antd';
import { api } from '../api';
import FolderList from '../components/FolderList';
import ItemTable from '../components/ItemTable';
import type { Folder, Item } from '../types';

export default function Overview() {
  const [activeFolder, setActiveFolder] = useState<number | null>(null);
  const [page, setPage] = useState(1);
  const [query, setQuery] = useState('');
  const [searching, setSearching] = useState(false);

  const { data: folders } = useRequest(() => api<{ folders: Folder[] }>('/api/folders'));
  const { data: folderItems, loading } = useRequest(
    () => activeFolder ? api<{ items: Item[]; total: number }>(`/api/folders/${activeFolder}/items?page=${page}&pageSize=20`) : Promise.resolve({ items: [], total: 0 }),
    { refreshDeps: [activeFolder, page] },
  );
  const { data: searchRes } = useRequest(
    () => query ? api<{ items: Item[] }>(`/api/items/search?q=${encodeURIComponent(query)}`) : Promise.resolve(null),
    { refreshDeps: [query] },
  );

  const displayItems = searching && searchRes ? searchRes.items : folderItems?.items ?? [];

  return (
    <div style={{ display: 'flex', gap: 16 }}>
      <FolderList folders={folders?.folders ?? []} active={activeFolder} onSelect={(id) => { setActiveFolder(id); setPage(1); setSearching(false); }} />
      <div style={{ flex: 1 }}>
        <Input
          prefix={<Search size={14} />}
          placeholder="搜索标题/简介/UP(跨全部收藏夹)"
          value={query}
          onChange={(e) => { setQuery(e.target.value); setSearching(e.target.value.length > 0); }}
          allowClear
          style={{ marginBottom: 12 }}
        />
        <ItemTable
          items={displayItems}
          total={searching ? (searchRes?.items.length ?? 0) : (folderItems?.total ?? 0)}
          page={page}
          pageSize={20}
          onPage={setPage}
        />
      </div>
    </div>
  );
}
```
